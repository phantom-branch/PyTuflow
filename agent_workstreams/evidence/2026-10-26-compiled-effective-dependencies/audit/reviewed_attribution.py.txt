 start=time.process_time()

def timed(label,original):
 def call(*a,**k):
  previous=phase[0];phase[0]=label;start=time.process_time()
  try:return original(*a,**k)
  finally:seconds[label]+=time.process_time()-start;phase[0]=previous
 return call

def scope(self,scopes):
 kinds={str(s.type).split(' (')[0] for s in scopes or ()}
 category=('variable' if 'VARIABLE' in kinds else 'combined' if 'SCENARIO' in kinds and kinds & {'EVENT','EVENT DEFINE'} else 'scenario' if 'SCENARIO' in kinds else 'event' if kinds & {'EVENT','EVENT DEFINE'} else 'other')
 counts[f'{phase[0]}.{category}.scope_calls']+=1
 return orig_scope(self,scopes)

def key(self,inp,ctx):
 kind={'geometry control file':'TGC','bc control file':'TBC','event file':'TEF'}.get(str(inp.lhs).lower(),'other')
 start=time.process_time();value=orig_key(self,inp,ctx);seconds['signatures']+=time.process_time()-start
 hit=value in self.cache or value in self.pending
 file_counts[kind]['hits' if hit else 'misses']+=1
 file_counts[kind]['compact' if value[1][0]=='manifest' else 'full']+=1
 return value

 'compiled_scope_fallbacks':getattr(getattr(plan,'scope_program',None),'fallbacks',0)}
